import pandas as pd
import networkx as nx
import matplotlib.pyplot as plt

# Load dataset
file_path = 
df = pd.read_csv(file_path)

# Reduce dataset for better visualization
df = df.sample(300, random_state=42)

# Create a graph
G = nx.Graph()

# Ensure both customers & merchants are added
customers = set(df["Account_Number"])  # Unique customer IDs
merchants = set(df["Merchant_Name"])   # Unique merchant names

# Add customers & merchants as nodes
for customer in customers:
    G.add_node(f"C_{customer}", type="Customer")

for merchant in merchants:
    G.add_node(f"M_{merchant}", type="Merchant")

# Add transaction edges
for index, row in df.iterrows():
    customer = f"C_{row['Account_Number']}"
    merchant = f"M_{row['Merchant_Name']}"
    G.add_edge(customer, merchant, weight=row["Transaction_Amount"])

# Try a different layout for better node spacing
pos = nx.kamada_kawai_layout(G)

# Separate Customers & Merchants
customers = [node for node, data in G.nodes(data=True) if data["type"] == "Customer"]
merchants = [node for node, data in G.nodes(data=True) if data["type"] == "Merchant"]

# Plot network
plt.figure(figsize=(12, 8))

# Draw customers (blue) and merchants (red)
nx.draw_networkx_nodes(G, pos, nodelist=customers, node_size=30, node_color="blue", alpha=0.7, label="Customers")
nx.draw_networkx_nodes(G, pos, nodelist=merchants, node_size=50, node_color="red", alpha=0.7, label="Merchants")
nx.draw_networkx_edges(G, pos, alpha=0.5, width=0.5)

plt.title("Banking Transactions Network Map")
plt.legend()
plt.show()
